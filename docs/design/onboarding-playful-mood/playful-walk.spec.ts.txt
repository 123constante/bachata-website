async function answer(page: Page, at: string) {
  const t = (id: string) => page.getByTestId(id);
  switch (at) {
    case 'contact-phone': return t('p3-ob-phone').fill('07700 900123');
    case 'whatsapp': return t('p3-ob-wa-no').click();
    case 'cat': return t('p3-ob-cat-Promoter').click();
    case 'link': return t('p3-ob-link').fill('@ritmo');
    case 'logo':
      await t('p3-ob-logo-file').setInputFiles({ name: 'l.png', mimeType: 'image/png', buffer: PNG });
      return expect(t('p3-ob-logo')).toHaveAttribute('data-has', 'true');
    case 'name': return t('p3-ob-name').fill('Sam');
    case 'photo':
      await t('p3-ob-photo-file').setInputFiles({ name: 'me.png', mimeType: 'image/png', buffer: PNG });
      return expect(t('p3-ob-ava')).toHaveAttribute('data-has', 'true');
    case 'role': return t('p3-ob-role-Leader').click();
    case 'nat': return t('p3-ob-nat-GB').click();
    case 'teach': return t('p3-ob-teach-yes').click();
    case 'levels': return t('p3-ob-level-Beginner').click();
    case 'styles': return t('p3-ob-styles-Moderna').click();
    case 'years': return t('p3-ob-years').fill('3');
  }
}


const WANT = new Set(['welcome', 'photo', 'nat', 'styles']);
for (const theme of ['light', 'dark'] as const) {
  test(`playful walk ${theme}`, async ({ page }) => {
    test.setTimeout(180_000);
    fs.mkdirSync(OUT, { recursive: true });
    await page.setViewportSize({ width: 390, height: 844 });
    await page.addInitScript((t) => { try { window.localStorage.setItem('p3-ob-theme', t); } catch { /* */ } }, theme);
    await setup(page, 'empty');
    if (process.env.FLAGS_FAIL) { await page.unroute('https://flagcdn.com/**'); await page.route('https://flagcdn.com/**', (r) => r.abort()); }
    await page.goto(`/account/p3/start?o=${ORG}`);
    await page.getByTestId('p3-ob-title').waitFor({ timeout: 20_000 });
    const seen = new Set<string>();
    const metrics: unknown[] = [];
    for (let i = 0; i < 30; i++) {
      const at = (await page.getByTestId('p3-ob-shell').getAttribute('data-screen')) ?? '';
      if ((await page.getByTestId('p3-ob-continue').getAttribute('aria-disabled')) === 'true') await answer(page, at);
      if (!seen.has(at)) {
        seen.add(at);
        await page.waitForTimeout(500);
        const m = await page.evaluate(() => {
          const main = document.querySelector('.p3-ob-main') as HTMLElement;
          const small: string[] = [];
          for (const el of Array.from(document.querySelectorAll('.p3-ob button, .p3-ob input, .p3-ob [role=radio]'))) {
            const b = el.getBoundingClientRect();
            if (b.width && (b.height < 43.5 || b.width < 43.5) && !(el as HTMLElement).classList.contains('p3-ob-sr')) small.push(`${el.className} ${Math.round(b.width)}x${Math.round(b.height)}`);
          }
          return { title: document.querySelector('[data-testid="p3-ob-title"]')?.textContent, scroll: main.scrollHeight - main.clientHeight, wide: document.documentElement.scrollWidth - innerWidth, small, flagBoxEmpty: Array.from(document.querySelectorAll('.p3-ob-flag')).filter((f) => !(f as HTMLElement).textContent && (f as HTMLImageElement).complete === false).length };
        });
        metrics.push({ screen: at, ...m });
        if (WANT.has(at)) await page.screenshot({ path: `${OUT}/${theme}-${at}.png` });
      }
      if (at === 'nat') { if ((await page.getByTestId('p3-ob-nat-GB').getAttribute('aria-checked')) !== 'true') await page.getByTestId('p3-ob-nat-GB').click(); await page.waitForTimeout(400); await page.screenshot({ path: `${OUT}/${theme}-nat-picked.png` }); }
      if (at === 'styles') { if ((await page.getByTestId('p3-ob-styles-Moderna').getAttribute('aria-pressed')) !== 'true') await page.getByTestId('p3-ob-styles-Moderna').click(); await page.waitForTimeout(400); await page.screenshot({ path: `${OUT}/${theme}-styles-picked.png` }); break; }
      if ((await page.getByTestId('p3-ob-continue').getAttribute('aria-disabled')) === 'true') await answer(page, at);
      await page.getByTestId('p3-ob-continue').click();
      await expect(page.getByTestId('p3-ob-shell')).not.toHaveAttribute('data-screen', at);
    }
    fs.writeFileSync(`${OUT}/${theme}-metrics.json`, JSON.stringify(metrics, null, 1));
  });
