// 1. No-access: every event-workspace screen with admin_event_workspace_p5 refused.
const NOACCESS = [`events/${FRIDAY}`, `events/${FRIDAY}/details`, `cover/${FRIDAY}`, `cancel/${FRIDAY}`, `cancel/${FRIDAY}/${OCC}`, `add/${FRIDAY}`, `add/${FRIDAY}/2026-10-16`, `change/${FRIDAY}/${OCC}/time`, `new/${FRIDAY}/date`];

    await page.screenshot({ path: `${OUT}/${slug}.png` });

    const back = page.locator('a:visible,button:visible').filter({ hasText: /^\s*(Back|Events|Event|Home)\s*$/ }).first();
    const backName = (await back.innerText().catch(() => '')).trim() || (await back.getAttribute('aria-label').catch(() => ''));

    log({ item: 1, s, backName, noAccess, retry, hasBack, before, after: new URL(page.url()).pathname, t: t.slice(0, 200), errors });

}

// 8. Focus return after closing a sheet (real click, then Escape / Close).
test('focus return', async ({ page }) => {
  await openOrganiser(page, `/account/o/events/${FRIDAY}/details`);
  await settle(page);
  const opener = page.locator('[data-testid="org-detail-venue"]').first();
  const openerLabel = (await opener.innerText()).replace(/\s+/g, ' ').slice(0, 40);
  await opener.click();
  await page.waitForTimeout(700);
  const sheetOpen = await page.locator('[role=dialog]').count();
  await page.keyboard.press('Escape');
  await page.waitForTimeout(700);
  const afterEsc = await page.evaluate(() => { const a = document.activeElement as HTMLElement | null; return a ? `${a.tagName} ${(a.innerText || a.getAttribute('aria-label') || '').replace(/\s+/g, ' ').slice(0, 40)}` : 'null'; });
  await page.evaluate((p) => { history.pushState({}, '', p); dispatchEvent(new PopStateEvent('popstate')); }, '/account/o/more');
  await settle(page);
  await page.getByText('Sign out', { exact: true }).click();
  await page.waitForTimeout(700);
  await page.screenshot({ path: `${OUT}/focus__default__signout_sheet.png` });
  const close = page.locator('[role=dialog]').getByRole('button', { name: /cancel|close|not now|stay/i }).first();
  const closeLabel = await close.innerText().catch(() => 'none');
  if (await close.count()) await close.click(); else await page.keyboard.press('Escape');
  await page.waitForTimeout(700);
  const dialogsAfter = await page.locator('[role=dialog]').count();
  const afterSignout = await page.evaluate(() => { const a = document.activeElement as HTMLElement | null; return a ? `${a.tagName} ${(a.innerText || a.getAttribute('aria-label') || '').replace(/\s+/g, ' ').slice(0, 40)}` : 'null'; });
  log({ item: 8, dialogsAfter, openerLabel, sheetOpen, afterEsc, closeLabel, afterSignout });
});
