async function answer(page: Page, at: string) {
  const t = (id: string) => page.getByTestId(id);
  switch (at) {
    case 'contact-phone': return t('p3-ob-phone').fill('07700 900123');
    case 'whatsapp': return t('p3-ob-wa-no').click();
    case 'cat': return t('p3-ob-cat-Promoter').click();
    case 'link': return t('p3-ob-link').fill('@ritmo');
    case 'logo':
      await t('p3-ob-logo-file').setInputFiles({ name: 'l.png', mimeType: 'image/png', buffer: PNG });
      return expect(t('p3-ob-logo')).toHaveAttribute('data-has', 'true');
    case 'name': return t('p3-ob-name').fill('Sam');
    case 'photo':
      await t('p3-ob-photo-file').setInputFiles({ name: 'me.png', mimeType: 'image/png', buffer: PNG });
      return expect(t('p3-ob-ava')).toHaveAttribute('data-has', 'true');
    case 'role': return t('p3-ob-role-Leader').click();
    case 'nat': return t('p3-ob-nat-GB').click();
    case 'teach': return t('p3-ob-teach-yes').click();
    case 'levels': return t('p3-ob-level-Beginner').click();
    case 'styles': return t('p3-ob-styles-Moderna').click();
    case 'years': return t('p3-ob-years').fill('3');
  }
}

async function record(page: Page, label: string) {
  await page.waitForLoadState('networkidle').catch(() => undefined); await page.waitForTimeout(450);
  const r = await page.evaluate(() => {
    const shell = document.querySelector('.p3-ob') as HTMLElement;
    const main = document.querySelector('.p3-ob-main') as HTMLElement;
    const q = (s: string) => document.querySelector(s) as HTMLElement | null;
    const small: string[] = [];
    for (const el of Array.from(shell.querySelectorAll('button, input, textarea, a, [role="radio"], [role="checkbox"]'))) {
      const b = el.getBoundingClientRect();
      if (!b.width || (el as HTMLElement).classList.contains('p3-ob-sr')) continue;
      if (b.height < 43.5 || b.width < 43.5) small.push(`${el.tagName}.${(el as HTMLElement).className} ${Math.round(b.width)}x${Math.round(b.height)}`);
    }
    const title = q('[data-testid="p3-ob-title"]');
    const btn = q('[data-testid="p3-ob-continue"]');
    const cs = (e: HTMLElement | null) => (e ? getComputedStyle(e) : null);
    const prog = q('[data-testid="p3-ob-progress"]');
    return {
      screen: shell.getAttribute('data-screen'),
      title: title?.textContent ?? null,
      text: (main?.innerText ?? '').replace(/\s+/g, ' ').slice(0, 400),
      pct: q('[data-testid="p3-ob-pct"]')?.textContent ?? null,
      pctVisible: !!q('[data-testid="p3-ob-pct"]')?.getBoundingClientRect().width,
      aria: prog?.getAttribute('aria-valuenow') ?? null,
      flags: shell.querySelectorAll('img.p3-ob-flag, .p3-ob-flag').length,
      inputs: shell.querySelectorAll('input:not([type=file]), textarea').length,
      cards: shell.querySelectorAll('[role=radio], [role=checkbox]').length,
      scroll: main.scrollHeight - main.clientHeight, shellScroll: shell.scrollHeight - shell.clientHeight,
      wide: document.documentElement.scrollWidth - innerWidth,
      small,
      titleColor: cs(title)?.color, titleWeight: cs(title)?.fontWeight, titleSize: cs(title)?.fontSize,
      btnRadius: cs(btn)?.borderRadius, btnBg: cs(btn)?.backgroundColor,
      emoji: /\p{Extended_Pictographic}/u.test(shell.innerText),
      confetti: shell.querySelectorAll('[class*=confetti], [class*=spark]').length,
    };
  });
  rows.push({ label, ...r });
  await page.screenshot({ path: `${OUT}/${String(rows.length).padStart(2, '0')}-${label}.png` });
}

test.afterAll(() => fs.writeFileSync(`${OUT}/rows.json`, JSON.stringify(rows, null, 1)));

test('walk empty persona to review, then ready review -> live', async ({ page }) => {
  test.setTimeout(180_000);
  await page.setViewportSize({ width: 390, height: 844 });
  await setup(page, 'empty');
  await page.goto(`/account/p3/start?o=${ORG}`);
  await page.waitForLoadState('networkidle').catch(() => undefined);
  const mounted = await page.getByTestId('p3-ob-title').waitFor({ timeout: 20_000 }).then(() => true, () => false);
  test.skip(!mounted, 'organiserP3 off');
  const seen = new Set<string>();
  for (let i = 0; i < 30; i++) {
    const at = (await page.getByTestId('p3-ob-shell').getAttribute('data-screen')) ?? '';
    if (!seen.has(at)) { seen.add(at); await record(page, at); }
    if (at === 'review' || at === 'live') break;
    if (at === 'nat') { await page.getByTestId('p3-ob-more').click(); await page.waitForTimeout(500); await page.screenshot({ path: `${OUT}/nat-sheet.png` }); await page.keyboard.press('Escape'); }
    if ((await page.getByTestId('p3-ob-continue').getAttribute('aria-disabled')) === 'true') await answer(page, at);
    await page.getByTestId('p3-ob-continue').click();
    await expect(page.getByTestId('p3-ob-shell')).not.toHaveAttribute('data-screen', at);
  }
  await page.unroute('**/rest/v1/rpc/organiser_onboarding_status_v1*');
  await page.route('**/rest/v1/rpc/organiser_onboarding_status_v1*', (r) =>
    json(r, { v: 1, gate_on: false, is_admin: false, complete: true, person: { complete: true, missing: [] },
      organisers: [{ id: ORG, name: ORG_NAME, role: 'owner', lifecycle: 'draft', complete: true, missing: [], about: { done: true, skipped: false } }] }));
  await page.goto(`/account/p3/start?o=${ORG}&at=review`);
  await page.waitForLoadState('networkidle').catch(() => undefined);
  await page.getByTestId('p3-ob-title').waitFor();
  await record(page, 'review-ready');
  await page.getByTestId('p3-ob-continue').click();
  await expect(page.getByTestId('p3-ob-shell')).toHaveAttribute('data-screen', 'live');
  await record(page, 'live');
});
