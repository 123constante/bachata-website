import { test, expect, type Page } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';
import { readFileSync, appendFileSync } from 'node:fs';
import { checkOverflow, fakeSupabase, frames } from './helpers';
import { FAKE_ORIGIN } from './fakeSupabase.mjs';

// THROWAWAY walk for the W2a design record (not committed): every shot at 390x844.
const SHAPES = JSON.parse(readFileSync(process.env.SHAPES_JSON!, 'utf8'));
const OUT = process.env.SHOTS!;
const sent: unknown[] = [];

async function open(page: Page, shape: string, scheme: 'light' | 'dark' = 'light', id = 'ser-1') {
  await page.emulateMedia({ colorScheme: scheme });
  await fakeSupabase(page, { signedInAsOwner: true, realClock: true });
  await page.clock.setFixedTime(new Date('2026-10-25T12:00:00Z'));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/admin_event_workspace_p5`, (r) =>
    shape === 'notfound'
      ? r.fulfill({ status: 400, contentType: 'application/json', body: JSON.stringify({ code: 'P0001', message: 'series_not_found: x' }) })
      : r.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify(SHAPES[shape]) }));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/series_command_p5`, (r) => { sent.push(r.request().postDataJSON()); return r.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ ok: true, new_version: 8 }) }); });
  await page.goto(`/account/p3/events/${id}`);
  await page.locator('[data-testid=p3-open-status],[data-testid=p3-event-notfound]').first().waitFor();
}

async function checks(page: Page, label: string, scope = '.p3-frame') {
  expect(await checkOverflow(page), `${label}: no sideways scroll`).toEqual([]);
  const small = await page.evaluate((s) => [...document.querySelectorAll(`${s} button, ${s} a[href], ${s} input`)]
    .map((el) => el.closest('label') ?? el).filter((el) => { const r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; })
    .map((el) => ({ el: (el.getAttribute('data-testid') ?? el.textContent ?? '').trim().slice(0, 40), h: Math.round(el.getBoundingClientRect().height) }))
    .filter((x) => x.h < 44), scope);
  expect(small, `${label}: taps >= 44px`).toEqual([]);
  const axe = await new AxeBuilder({ page }).include(scope).withRules(['color-contrast']).analyze();
  const clip = await page.evaluate((s) => [...document.querySelectorAll(`${s} *`)]
    .filter((el) => { const c = getComputedStyle(el); return el.children.length === 0 && (el.textContent ?? '').trim() && el.scrollWidth > el.clientWidth + 1 && c.overflow !== 'visible' && el.clientWidth > 0; })
    .map((el) => (el.textContent ?? '').trim().slice(0, 40)), scope);
  const taps = await page.evaluate((s) => [...document.querySelectorAll(`${s} button, ${s} a[href], ${s} input`)]
    .map((el) => (el.closest('label') ?? el).getBoundingClientRect()).filter((r) => r.width > 0 && r.height > 0).map((r) => Math.round(r.height)), scope);
  const contrast = await new AxeBuilder({ page }).include(scope).withRules(['color-contrast']).analyze();
  appendFileSync(`${OUT}/metrics.jsonl`, JSON.stringify({ label, sideways: (await checkOverflow(page)).length, taps: taps.length, minTap: Math.min(...taps), clipped: clip, contrastViolations: contrast.violations.length, contrastPasses: contrast.passes.flatMap((v) => v.nodes).length }) + '\n');
  expect(axe.violations.flatMap((v) => v.nodes.map((n) => n.target.join(' '))), `${label}: contrast`).toEqual([]);
}

const shot = async (page: Page, name: string) => { await frames(page); await page.screenshot({ path: `${OUT}/${name}.png` }); };
const sheet = async (page: Page) => { await page.getByTestId('p3-open-status').click(); await expect(page.getByTestId('p3-sheet')).toBeVisible(); };

test.beforeEach(({ page: _p }, info) => { test.skip(info.project.name !== 'phone-390x844', 'one size'); });

for (const scheme of ['light', 'dark'] as const) {
  test(`weekly page + status sheet ${scheme}`, async ({ page }) => {
    await open(page, 'weekly', scheme);
    await expect(page.getByTestId('p3-event-name')).toHaveText('Sensual Fridays');
    await checks(page, 'weekly page');
    await shot(page, `01-weekly-${scheme}`);
    await page.getByTestId('p3-coming').scrollIntoViewIfNeeded();
    await shot(page, `02-weekly-coming-about-${scheme}`);
    await sheet(page);
    await checks(page, 'weekly sheet', '.p3-sheet');
    await shot(page, `03-status-sheet-weekly-${scheme}`);
  });
}
