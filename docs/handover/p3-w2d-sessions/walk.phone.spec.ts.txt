/** organiser_get_occurrence_programme_v1 for one weekly date: 2 series sessions (+ an extra, a not-running one on 13 Nov). */
function dateProg(id: string, over: Record<string, unknown> = {}) {
  const date = id.slice(2);
  const special = date === '2026-11-13';
  const sessions: Record<string, unknown>[] = [
    { series_item_id: 'cls', type: 'class', title: 'Bachata class', start_time: '20:00:00', end_time: '21:00:00', ends_next_day: false, level_keys: ['open_level'], removed: special },
    { series_item_id: 'pty', type: 'party', title: 'Social', start_time: '21:00:00', end_time: '01:00:00', ends_next_day: true, level_keys: [], removed: false },
  ];
  if (special) sessions.push({ added_session_id: 'add1', type: 'performance', title: 'Guest show', start_time: '23:00:00', end_time: '23:20:00', ends_next_day: false, level_keys: [], removed: false });
  return {
    occurrence_id: id, series_id: 'ser-1', occurrence_date: date, version: 4,
    editable: date >= '2026-10-25' && date !== '2026-11-06', not_editable_reason: date === '2026-11-06' ? 'date_cancelled' : date < '2026-10-25' ? 'past_date' : null,
    sessions,
    session_people: [
      { series_item_id: 'cls', people: [{ profile_id: 'p-cls-0', display_name: 'Marco & Elena', role: 'teaching' }] },
      { series_item_id: 'pty', people: [{ profile_id: 'p-pty-0', display_name: 'DJ Kizzy', role: 'djing' }, { profile_id: 'p-pty-1', display_name: 'Host Sam', role: 'mc' }] },
      ...(special ? [{ added_session_id: 'add1', people: [] }] : []),
    ],
    ...over,
  };
}

async function open(page: Page, shape: string, scheme: 'light' | 'dark', progOver: Record<string, unknown> = {}) {
  await page.emulateMedia({ colorScheme: scheme });
  await fakeSupabase(page, { signedInAsOwner: true, realClock: true });
  await page.clock.setFixedTime(new Date('2026-10-25T12:00:00Z'));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/admin_event_workspace_p5`, (r) => r.fulfill(json(SHAPES[shape])));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/organiser_get_occurrence_programme_v1`, (r) => r.fulfill(json(dateProg(r.request().postDataJSON().p_occurrence_id, progOver))));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/organiser_search_people_v1`, (r) => r.fulfill(json([
    { id: 'n1', display_name: 'Maria Lopez', dj_name: null, photo_url: null, city_name: 'Leeds', country_code: 'GB' },
    { id: 'n2', display_name: 'Mario Rossi', dj_name: 'DJ Mario', photo_url: null, city_name: 'London', country_code: null }])));
  await page.route(`${FAKE_ORIGIN}/rest/v1/venue_rooms*`, (r) => r.fulfill(json([{ id: 'r1', name: 'Main room', sort_order: 0 }, { id: 'r2', name: 'Studio', sort_order: 1 }])));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/event_view_p5`, (r) => r.fulfill(json({ event: {}, occurrence: {}, schedule: {}, program: [], added_sessions: [] })));
  await page.route(`${FAKE_ORIGIN}/rest/v1/rpc/*_command_p5`, (r) => r.fulfill(json({ ok: true, new_version: 8 })));
  await page.goto('/account/p3/events/ser-1');
  await page.getByTestId('p3-open-status').waitFor();
}

async function checks(page: Page, label: string, scope = '.p3-frame') {
  expect(await checkOverflow(page), `${label}: no sideways scroll`).toEqual([]);
  const small = await page.evaluate((s) => [...document.querySelectorAll(`${s} button, ${s} a[href], ${s} input`)]
    .map((el) => el.closest('label') ?? el).filter((el) => { const r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; })
    .map((el) => ({ el: (el.getAttribute('data-testid') ?? el.textContent ?? '').trim().slice(0, 40), h: Math.round(el.getBoundingClientRect().height) }))
    .filter((x) => x.h < 44), scope);
  const clip = await page.evaluate((s) => [...document.querySelectorAll(`${s} *`)]
    .filter((el) => { const c = getComputedStyle(el); return el.children.length === 0 && (el.textContent ?? '').trim() && el.scrollWidth > el.clientWidth + 1 && c.overflow !== 'visible' && el.clientWidth > 0; })
    .map((el) => (el.textContent ?? '').trim().slice(0, 40)), scope);
  const taps = await page.evaluate((s) => [...document.querySelectorAll(`${s} button, ${s} a[href], ${s} input`)]
    .map((el) => (el.closest('label') ?? el).getBoundingClientRect()).filter((r) => r.width > 0 && r.height > 0).map((r) => Math.round(r.height)), scope);
  const contrast = await new AxeBuilder({ page }).include(scope).withRules(['color-contrast']).analyze();
  const sideways = (await checkOverflow(page)).length;
  appendFileSync(`${OUT}/metrics.jsonl`, JSON.stringify({ label, sideways, taps: taps.length, minTap: taps.length ? Math.min(...taps) : null, small, clipped: clip, contrastViolations: contrast.violations.flatMap((v) => v.nodes.map((n) => n.target.join(' '))), contrastPasses: contrast.passes.flatMap((v) => v.nodes).length }) + '\n');
  expect(small, `${label}: taps >= 44px`).toEqual([]);
  expect(clip, `${label}: clipped`).toEqual([]);
  expect(contrast.violations.flatMap((v) => v.nodes.map((n) => n.target.join(' '))), `${label}: contrast`).toEqual([]);
}
